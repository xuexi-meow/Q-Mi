# Q-Mi storage 子系统实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 Q-Mi 落地消息记忆库：收发消息、撤回标记、notice 存 SQLite，提供按 id / 最近 N 条 / 过滤 / 全文搜索查询。

**Architecture:** 新增 `storage` 包（不 import event/handler，只暴露自己的记录类型），`handler/store.go` 在事件到来时做结构体→记录转换并落库，`main.go` 启动时 Open 挂到 `app.App.Store`、退出时 Close。

**Tech Stack:** Go + `modernc.org/sqlite`（纯 Go SQLite 驱动，driver 名 `sqlite`）+ 现有 viper 配置。

**Spec:** `docs/superpowers/specs/2026-10-18-storage-design.md`（字段定义、表结构、API 以 spec 为准）

## Global Constraints

- 驱动：`modernc.org/sqlite`，driver 名 `sqlite`，禁止引入 CGO 驱动（mattn/go-sqlite3）。
- `storage` 包不得 import `Q-Mi/event`、`Q-Mi/handler`；事件→记录的转换只发生在 handler 层。
- 撤回只打标记（`recalled=1`），不删行。
- handler 的 `init()` 先于 main 打开数据库执行 → 落库 handler **每次事件到来时**从 `app.GetApp().Store` 取 Store，判 nil 即跳过，禁止在 init 里缓存。
- 事件分发是并发 goroutine，`Store` 必须并发安全（`db.SetMaxOpenConns(1)` + WAL）。
- 落库失败只 `slog.Warn`，绝不 panic / 影响聊天流程。
- 每个任务结束跑 `go test ./...` 全绿 + `go build ./...`。

```go
// Package storage 消息记忆库:收发消息、撤回标记、notice 事件的 SQLite 存取。
// 本包不 import event/handler,事件结构体到记录的转换在 handler 层完成。
package storage

import (
	"database/sql"
	"fmt"

	_ "modernc.org/sqlite"
)

const schema = `
CREATE TABLE IF NOT EXISTS messages (
    message_id  INTEGER PRIMARY KEY,
    direction   TEXT NOT NULL,
    chat_type   TEXT NOT NULL,
    chat_id     INTEGER NOT NULL,
    user_id     INTEGER NOT NULL,
    raw_message TEXT NOT NULL,
    segments    TEXT NOT NULL,
    time        REAL NOT NULL,
    recalled    INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX IF NOT EXISTS idx_messages_chat_time ON messages(chat_id, time DESC);
CREATE INDEX IF NOT EXISTS idx_messages_user_time ON messages(user_id, time DESC);

CREATE TABLE IF NOT EXISTS notices (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    notice_type TEXT NOT NULL,
    sub_type    TEXT,
    chat_id     INTEGER,
    user_id     INTEGER,
    raw         TEXT NOT NULL,
    time        REAL NOT NULL
);

CREATE VIRTUAL TABLE IF NOT EXISTS messages_fts
    USING fts5(raw_message, content='messages', content_rowid='message_id');
CREATE TRIGGER IF NOT EXISTS messages_ai AFTER INSERT ON messages BEGIN
    INSERT INTO messages_fts(rowid, raw_message) VALUES (new.message_id, new.raw_message);
END;
CREATE TRIGGER IF NOT EXISTS messages_ad AFTER DELETE ON messages BEGIN
    INSERT INTO messages_fts(messages_fts, rowid, raw_message)
    VALUES ('delete', old.message_id, old.raw_message);
END;
`

type Store struct {
	db *sql.DB
}

// Open 打开(必要时创建)数据库并执行幂等建表。
// path 传 ":memory:" 可得进程内库(测试用)。
func Open(path string) (*Store, error) {
	dsn := fmt.Sprintf("file:%s?_pragma=journal_mode(WAL)&_pragma=busy_timeout(5000)", path)
	db, err := sql.Open("sqlite", dsn)
	if err != nil {
		return nil, fmt.Errorf("打开数据库失败: %w", err)
	}
	// modernc 驱动多连接写会撞锁,单连接 + WAL 足够本项目的写入量
	db.SetMaxOpenConns(1)
	if _, err := db.Exec(schema); err != nil {
		db.Close()
		return nil, fmt.Errorf("建表失败: %w", err)
	}
	return &Store{db: db}, nil
}

func (s *Store) Close() error {
	return s.db.Close()
}
```

```go
package storage

import "fmt"

// MessageRecord messages 表的一行
type MessageRecord struct {
	MessageID  int64
	Direction  string // "recv" / "sent"
	ChatType   string // "private" / "group"
	ChatID     int64  // 群号;私聊 = 对方 QQ
	UserID     int64  // 发送者;发出 = 机器人自身
	RawMessage string
	Segments   []byte // 原始消息段 JSON
	Time       float64
	Recalled   bool
}

const messageCols = "message_id, direction, chat_type, chat_id, user_id, raw_message, segments, time, recalled"

func (s *Store) PutMessage(r *MessageRecord) error {
	recalled := 0
	if r.Recalled {
		recalled = 1
	}
	_, err := s.db.Exec(`
INSERT INTO messages (message_id, direction, chat_type, chat_id, user_id, raw_message, segments, time, recalled)
VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
ON CONFLICT(message_id) DO UPDATE SET
    direction=excluded.direction, chat_type=excluded.chat_type, chat_id=excluded.chat_id,
    user_id=excluded.user_id, raw_message=excluded.raw_message, segments=excluded.segments,
    time=excluded.time, recalled=excluded.recalled`,
		r.MessageID, r.Direction, r.ChatType, r.ChatID, r.UserID, r.RawMessage, string(r.Segments), r.Time, recalled)
	if err != nil {
		return fmt.Errorf("写入消息失败: %w", err)
	}
	return nil
}

func (s *Store) GetMessage(id int64) (*MessageRecord, error) {
	row := s.db.QueryRow("SELECT "+messageCols+" FROM messages WHERE message_id = ?", id)
	r, err := scanMessage(row)
	if err != nil {
		if err == sql.ErrNoRows {
			return nil, nil
		}
		return nil, fmt.Errorf("查询消息 %d 失败: %w", id, err)
	}
	return r, nil
}

// scanner 抽象 *sql.Row 与 *sql.Rows,共用一个 scan 逻辑
type scanner interface{ Scan(dest ...any) error }

func scanMessage(row scanner) (*MessageRecord, error) {
	var r MessageRecord
	var segments []byte
	var recalled int
	if err := row.Scan(&r.MessageID, &r.Direction, &r.ChatType, &r.ChatID, &r.UserID,
		&r.RawMessage, &segments, &r.Time, &recalled); err != nil {
		return nil, err
	}
	r.Segments = segments
	r.Recalled = recalled != 0
	return &r, nil
}
```

```go
// RecentMessages 某 chat 最近 N 条,按时间倒序,已撤回默认排除
func (s *Store) RecentMessages(chatType string, chatID int64, limit int) ([]*MessageRecord, error) {
	return s.QueryMessages(MessageFilter{ChatType: chatType, ChatID: chatID, Limit: limit})
}

// MessageFilter 查询过滤器,零值字段表示不过滤
type MessageFilter struct {
	ChatType        string
	ChatID, UserID  int64
	Since, Until    time.Time
	IncludeRecalled bool
	Limit           int
}

func (s *Store) QueryMessages(f MessageFilter) ([]*MessageRecord, error) {
	if f.Limit <= 0 {
		f.Limit = 50
	}
	where := []string{"1=1"}
	var args []any
	if f.ChatType != "" {
		where = append(where, "chat_type = ?")
		args = append(args, f.ChatType)
	}
	if f.ChatID != 0 {
		where = append(where, "chat_id = ?")
		args = append(args, f.ChatID)
	}
	if f.UserID != 0 {
		where = append(where, "user_id = ?")
		args = append(args, f.UserID)
	}
	if !f.Since.IsZero() {
		where = append(where, "time >= ?")
		args = append(args, float64(f.Since.UnixMilli())/1000)
	}
	if !f.Until.IsZero() {
		where = append(where, "time < ?")
		args = append(args, float64(f.Until.UnixMilli())/1000)
	}
	if !f.IncludeRecalled {
		where = append(where, "recalled = 0")
	}
	args = append(args, f.Limit)

	rows, err := s.db.Query("SELECT "+messageCols+" FROM messages WHERE "+
		strings.Join(where, " AND ")+" ORDER BY time DESC LIMIT ?", args...)
	if err != nil {
		return nil, fmt.Errorf("查询消息失败: %w", err)
	}
	defer rows.Close()

	var out []*MessageRecord
	for rows.Next() {
		r, err := scanMessage(rows)
		if err != nil {
			return nil, fmt.Errorf("扫描消息行失败: %w", err)
		}
		out = append(out, r)
	}
	return out, rows.Err()
}

// MarkRecalled 撤回只打标记不删行,重复标记幂等
func (s *Store) MarkRecalled(id int64) error {
	_, err := s.db.Exec("UPDATE messages SET recalled = 1 WHERE message_id = ?", id)
	if err != nil {
		return fmt.Errorf("标记撤回失败: %w", err)
	}
	return nil
}
```

```go
// SearchMessages FTS5 全文搜索(中文按字索引),按时间倒序
func (s *Store) SearchMessages(query string, limit int) ([]*MessageRecord, error) {
	if limit <= 0 {
		limit = 50
	}
	rows, err := s.db.Query(`
SELECT m.message_id, m.direction, m.chat_type, m.chat_id, m.user_id,
       m.raw_message, m.segments, m.time, m.recalled
FROM messages_fts f JOIN messages m ON m.message_id = f.rowid
WHERE messages_fts MATCH ?
ORDER BY m.time DESC LIMIT ?`, query, limit)
	if err != nil {
		return nil, fmt.Errorf("搜索消息失败: %w", err)
	}
	defer rows.Close()

	var out []*MessageRecord
	for rows.Next() {
		r, err := scanMessage(rows)
		if err != nil {
			return nil, fmt.Errorf("扫描搜索结果失败: %w", err)
		}
		out = append(out, r)
	}
	return out, rows.Err()
}

// NoticeRecord notices 表的一行
type NoticeRecord struct {
	NoticeType string
	SubType    string
	ChatID     int64
	UserID     int64
	Raw        []byte
	Time       float64
}

func (s *Store) PutNotice(n *NoticeRecord) error {
	_, err := s.db.Exec(`
INSERT INTO notices (notice_type, sub_type, chat_id, user_id, raw, time)
VALUES (?, ?, ?, ?, ?, ?)`,
		n.NoticeType, n.SubType, n.ChatID, n.UserID, string(n.Raw), n.Time)
	if err != nil {
		return fmt.Errorf("写入通知失败: %w", err)
	}
	return nil
}
```

```go
// store.go 事件落库:收发消息、撤回标记、其余 notice 原始 JSON 存档。
// Store 在 init 时可能尚未打开(main 在 handler init 之后才 Open),
// 所以每个 handler 都在事件到来时才从 app 取 Store,nil 即跳过。
package handler

import (
	"encoding/json"
	"log/slog"

	"Q-Mi/app"
	"Q-Mi/event"
	"Q-Mi/storage"
)

func init() {
	event.RegisterTyped("message.private", func(e *event.PrivateMessageEvent) {
		putMessage(privateRecord(e))
	})
	event.RegisterTyped("message.group", func(e *event.GroupMessageEvent) {
		putMessage(groupRecord(e))
	})
	event.RegisterTyped("message_sent.private", func(e *event.MessageSentEvent) {
		putMessage(sentRecord(e, "private"))
	})
	event.RegisterTyped("message_sent.group", func(e *event.MessageSentEvent) {
		putMessage(sentRecord(e, "group"))
	})

	event.RegisterTyped("notice.group_recall", func(e *event.GroupRecallEvent) {
		st := app.GetApp().Store
		if st == nil {
			return
		}
		if err := st.MarkRecalled(int64(e.MessageID)); err != nil {
			slog.Warn("撤回标记失败", "message_id", e.MessageID, "err", err)
		}
	})
	event.RegisterTyped("notice.friend_recall", func(e *event.FriendRecallEvent) {
		st := app.GetApp().Store
		if st == nil {
			return
		}
		if err := st.MarkRecalled(int64(e.MessageID)); err != nil {
			slog.Warn("撤回标记失败", "message_id", e.MessageID, "err", err)
		}
	})

	for _, key := range noticeStoreKeys {
		k := key
		event.Register(k, func(raw json.RawMessage) {
			storeNotice(k, raw)
		})
	}
}

// noticeStoreKeys 全部落库的 notice key(与 dispatcher 的 key 规则一致;
// 撤回两类不在此列,它们走 MarkRecalled)
var noticeStoreKeys = []string{
	"notice.group_upload", "notice.group_admin", "notice.group_decrease",
	"notice.group_increase", "notice.group_ban", "notice.friend_add",
	"notice.group_card", "notice.group_essence", "notice.group_msg_emoji_like",
	"notice.notify.group_name", "notice.notify.group_title",
	"notice.notify.poke", "notice.notify.profile_like", "notice.notify.input_status",
	"notice.bot_offline",
}

func putMessage(r *storage.MessageRecord) {
	st := app.GetApp().Store
	if st == nil {
		return
	}
	if err := st.PutMessage(r); err != nil {
		slog.Warn("消息落库失败", "message_id", r.MessageID, "err", err)
	}
}

func privateRecord(e *event.PrivateMessageEvent) *storage.MessageRecord {
	segs, _ := json.Marshal(e.Message) // Message.Parsed 是 json.RawMessage,重编即原始段 JSON
	return &storage.MessageRecord{
		MessageID: int64(e.MessageID), Direction: "recv", ChatType: "private",
		ChatID: int64(e.UserID), UserID: int64(e.UserID),
		RawMessage: e.RawMessage, Segments: segs, Time: e.Time,
	}
}

func groupRecord(e *event.GroupMessageEvent) *storage.MessageRecord {
	segs, _ := json.Marshal(e.Message)
	return &storage.MessageRecord{
		MessageID: int64(e.MessageID), Direction: "recv", ChatType: "group",
		ChatID: int64(e.GroupID), UserID: int64(e.UserID),
		RawMessage: e.RawMessage, Segments: segs, Time: e.Time,
	}
}

func sentRecord(e *event.MessageSentEvent, chatType string) *storage.MessageRecord {
	segs, _ := json.Marshal(e.Message)
	r := &storage.MessageRecord{
		MessageID: int64(e.MessageID), Direction: "sent", ChatType: chatType,
		UserID: int64(e.UserID), // 发出消息 user_id = 机器人自身
		RawMessage: e.RawMessage, Segments: segs, Time: e.Time,
	}
	if chatType == "group" {
		r.ChatID = int64(e.GroupID)
	} else {
		r.ChatID = int64(e.TargetID) // 私聊 chat_id = 对方 QQ
	}
	return r
}

func storeNotice(key string, raw json.RawMessage) {
	st := app.GetApp().Store
	if st == nil {
		return
	}
	var head struct {
		NoticeType string  `json:"notice_type"`
		SubType    string  `json:"sub_type"`
		GroupID    int     `json:"group_id"`
		UserID     int     `json:"user_id"`
		Time       float64 `json:"time"`
	}
	if err := json.Unmarshal(raw, &head); err != nil {
		slog.Warn("notice 解析失败,跳过落库", "key", key, "err", err)
		return
	}
	n := &storage.NoticeRecord{
		NoticeType: head.NoticeType, SubType: head.SubType,
		ChatID: int64(head.GroupID), UserID: int64(head.UserID),
		Raw: raw, Time: head.Time,
	}
	if err := st.PutNotice(n); err != nil {
		slog.Warn("notice 落库失败", "key", key, "err", err)
	}
}
```
